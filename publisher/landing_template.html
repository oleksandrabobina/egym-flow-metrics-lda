def generate_landing_page(realm_meta, latest_month_str): print("Generating Master Landing Page (index.html)...") landing_template = load_template("landing_template.html") flow_cards_html = "" incident_cards_html = "" cfr_cards_html = "" for realm_name, cfg in realm_meta.items(): # SECTION 1: FLOW METRICS # Points directly to your colleague's existing directory structure (e.g. apps/index.html) flow_cards_html += f'''

{cfg['icon']} {realm_name}

{cfg['teams']} teams · Latest: {latest_month_str}

Open Dashboard →
''' # SECTION 2: DORA INCIDENTS (PART 1) # Points strictly to your generated public/ files (e.g. public/apps.html) incident_cards_html += f'''

{cfg['icon']} {realm_name}

{cfg['teams']} teams · Latest: {latest_month_str}

Open Dashboard →
''' # SECTION 3: DORA CFR (PART 2) # Points strictly to your generated public/ CFR files (e.g. public/apps-cfr-dashboard.html) if cfg["cfr"]: cfr_cards_html += f'''

{cfg['icon']} {realm_name}

{cfg['teams']} teams · Latest: {latest_month_str}

Open Dashboard →
''' landing_html = landing_template.replace("{{FLOW_CARDS}}", flow_cards_html) landing_html = landing_html.replace("{{INCIDENT_CARDS}}", incident_cards_html) landing_html = landing_html.replace("{{CFR_CARDS}}", cfr_cards_html) with open("index.html", "w", encoding="utf-8") as f: f.write(landing_html) print("Master Landing Page index.html Generated Successfully!")
